cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PacketiaVideoTranscoderTests LANGUAGES CXX)
    enable_testing()
    find_package(Threads REQUIRED)
    find_package(GTest REQUIRED)
    get_filename_component(PACKETIA_TRANSCODE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    include("${PACKETIA_TRANSCODE_ROOT}/utils/threading/ThreadSupport.cmake")
    add_subdirectory("${PACKETIA_TRANSCODE_ROOT}/Common/memory" "${CMAKE_CURRENT_BINARY_DIR}/memory")
    add_subdirectory("${PACKETIA_TRANSCODE_ROOT}/cmake/ffmpeg" "${CMAKE_CURRENT_BINARY_DIR}/ffmpeg")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/transcoder")
endif()

add_executable(test_transcode_domain test_transcode_domain.cpp)
target_link_libraries(test_transcode_domain PRIVATE packetia_transcode_domain GTest::gtest_main)
add_test(NAME transcode_domain COMMAND test_transcode_domain)

add_executable(test_video_transcoder test_video_transcoder.cpp)
target_link_libraries(test_video_transcoder PRIVATE packetia_transcode_service GTest::gtest_main)
if(TARGET media)
    target_compile_definitions(test_video_transcoder PRIVATE PACKETIA_TEST_SFU_PIPELINE)
else()
    target_sources(test_video_transcoder PRIVATE
        ../../../media/core/EncodedFrameRouter.cpp)
    target_include_directories(test_video_transcoder PRIVATE ../../../log)
endif()
add_test(NAME video_transcoder COMMAND test_video_transcoder)
set_tests_properties(video_transcoder PROPERTIES TIMEOUT 90)

add_executable(test_multicodec test_multicodec.cpp)
target_link_libraries(test_multicodec PRIVATE packetia_transcode_service GTest::gtest_main)
if(NOT TARGET media)
    target_sources(test_multicodec PRIVATE ../../../media/core/EncodedFrameRouter.cpp)
    target_include_directories(test_multicodec PRIVATE ../../../log)
endif()
add_test(NAME multicodec COMMAND test_multicodec)
set_tests_properties(multicodec PROPERTIES TIMEOUT 120)
