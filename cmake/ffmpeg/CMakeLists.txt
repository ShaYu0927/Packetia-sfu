if(TARGET packetia_ffmpeg)
    return()
endif()

# One installation provides both codec/container libraries and the scaler.
# A custom prefix is strict: missing packages must never fall back to system FFmpeg.
set(PACKETIA_FFMPEG_ROOT "" CACHE PATH "FFmpeg installation prefix; empty uses system packages")
find_package(PkgConfig REQUIRED)
if(NOT "${_PACKETIA_FFMPEG_PREVIOUS_ROOT}" STREQUAL "${PACKETIA_FFMPEG_ROOT}")
    unset(__pkg_config_checked_PACKETIA_FFMPEG CACHE)
    unset(__pkg_config_arguments_PACKETIA_FFMPEG CACHE)
    get_cmake_property(_packetia_cached_variables CACHE_VARIABLES)
    foreach(variable IN LISTS _packetia_cached_variables)
        if(variable MATCHES "^pkgcfg_lib_PACKETIA_FFMPEG_")
            unset(${variable} CACHE)
        endif()
    endforeach()
endif()
set(_PACKETIA_FFMPEG_PREVIOUS_ROOT "${PACKETIA_FFMPEG_ROOT}" CACHE INTERNAL "Previous FFmpeg prefix")

if(PACKETIA_FFMPEG_ROOT)
    get_filename_component(_packetia_ffmpeg_root "${PACKETIA_FFMPEG_ROOT}" REALPATH)
    foreach(variable PKG_CONFIG_PATH PKG_CONFIG_LIBDIR)
        if(DEFINED ENV{${variable}})
            set(_packetia_had_${variable} TRUE)
            set(_packetia_saved_${variable} "$ENV{${variable}}")
        endif()
    endforeach()
    if(WIN32)
        set(_packetia_pc_separator ";")
    else()
        set(_packetia_pc_separator ":")
    endif()
    set(_packetia_pc_path "${_packetia_ffmpeg_root}/lib/pkgconfig${_packetia_pc_separator}${_packetia_ffmpeg_root}/lib64/pkgconfig")
    set(ENV{PKG_CONFIG_PATH} "${_packetia_pc_path}")
    set(ENV{PKG_CONFIG_LIBDIR} "${_packetia_pc_path}")
endif()

if(PACKETIA_FFMPEG_ROOT)
    set(_packetia_pkg_options NO_CMAKE_PATH NO_CMAKE_ENVIRONMENT_PATH)
endif()
pkg_check_modules(PACKETIA_FFMPEG REQUIRED IMPORTED_TARGET GLOBAL
    ${_packetia_pkg_options} libavformat libavcodec libavutil libswscale)

if(PACKETIA_FFMPEG_ROOT)
    foreach(package libavformat libavcodec libavutil libswscale)
        pkg_get_variable(_packetia_package_prefix ${package} prefix)
        get_filename_component(_packetia_package_prefix "${_packetia_package_prefix}" REALPATH)
        if(NOT _packetia_package_prefix STREQUAL _packetia_ffmpeg_root)
            message(FATAL_ERROR "${package} must come from PACKETIA_FFMPEG_ROOT, found ${_packetia_package_prefix}")
        endif()
    endforeach()
    foreach(library IN LISTS PACKETIA_FFMPEG_LINK_LIBRARIES)
        get_filename_component(_packetia_library_name "${library}" NAME)
        if(_packetia_library_name MATCHES "^(lib)?(avformat|avcodec|avutil|swscale|swresample)([.-]|$)")
            if(NOT IS_ABSOLUTE "${library}")
                message(FATAL_ERROR "FFmpeg library ${library} is missing from PACKETIA_FFMPEG_ROOT")
            endif()
            get_filename_component(_packetia_library_realpath "${library}" REALPATH)
            string(FIND "${_packetia_library_realpath}" "${_packetia_ffmpeg_root}/" _packetia_prefix_position)
            if(NOT _packetia_prefix_position EQUAL 0)
                message(FATAL_ERROR "FFmpeg library ${library} is outside PACKETIA_FFMPEG_ROOT; refusing mixed installations")
            endif()
        endif()
    endforeach()
    foreach(variable PKG_CONFIG_PATH PKG_CONFIG_LIBDIR)
        if(_packetia_had_${variable})
            set(ENV{${variable}} "${_packetia_saved_${variable}}")
        else()
            unset(ENV{${variable}})
        endif()
    endforeach()
    message(STATUS "Packetia FFmpeg: ${_packetia_ffmpeg_root} (avcodec ${PACKETIA_FFMPEG_libavcodec_VERSION})")
else()
    message(STATUS "Packetia FFmpeg: system packages (avcodec ${PACKETIA_FFMPEG_libavcodec_VERSION})")
endif()

add_library(packetia_ffmpeg INTERFACE)
target_link_libraries(packetia_ffmpeg INTERFACE PkgConfig::PACKETIA_FFMPEG)

set(FFMPEG_INCLUDE_DIRS "${PACKETIA_FFMPEG_INCLUDE_DIRS}" PARENT_SCOPE)
set(FFMPEG_LIBRARIES packetia_ffmpeg PARENT_SCOPE)
